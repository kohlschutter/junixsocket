133   public void testFileChannelMMap() throws Exception {
134     String name = "juxtest.FileChannelMap";
135 
136     try {
137       SharedMemory.unlinkShared(name); // forcibly remove, just in case
138       try (SharedMemory mem = SharedMemory.createOrReuseExisting(name, 3,
139           SharedMemoryOption.UNLINK_UPON_CLOSE)) {
140 
141         int actuallyAllocatedSize = (int) mem.byteSize() /* (int) fc.size() */;
142         // System.out.println("Allocation size: " + actuallyAllocatedSize);
143         assertTrue(actuallyAllocatedSize >= 3);
144 
145         try (FileChannel fc = mem.asMappableFileChannel()) {
146           assertEquals(actuallyAllocatedSize, fc.size());
147           MappedByteBuffer mbb = fc.map(MapMode.READ_WRITE, 0, actuallyAllocatedSize);
148           assertTrue(mbb.capacity() >= 3);
149           assertTrue(mbb.limit() >= 3);
150           assertEquals(0, mbb.position());
151 
152           mbb.put((byte) 1);
153           mbb.put((byte) 2);
154           mbb.put((byte) 3);
155           mbb.put(actuallyAllocatedSize - 1, (byte) 3);
156 
157           int actuallyAllocatedSize0 = (int) fc.size(); // don't expect any extra allocations for
158                                                         // that
159                                                         // fourth put
160           assertEquals(actuallyAllocatedSize, actuallyAllocatedSize0);
161 
162           assertThrows(FileAlreadyExistsException.class, () -> SharedMemory.createExclusively(name,
163               16));
164 
165           // mbb.force();
166 
167           // opening an existing mapping by name should retain its data
168           try (SharedMemory mem1 = SharedMemory.openExisting(name);
169               FileChannel fc1 = mem1.asMappableFileChannel()) {
170             int actuallyAllocatedSize1 = (int) fc1.size();
171             assertEquals(actuallyAllocatedSize0, actuallyAllocatedSize1);
172 
173             MappedByteBuffer mbb1 = fc1.map(MapMode.READ_WRITE, 0, actuallyAllocatedSize1);
174             assertEquals(1, mbb1.get());
175             assertEquals(2, mbb1.get());
176             assertEquals(3, mbb1.get());
177 
178             assertFalse(mbb1.isReadOnly());
179           }
180 
181           // reusing an existing mapping should still clear out the previous memory
182           try (SharedMemory mem1 = SharedMemory.createOrReuseExisting(name, 3);
183               FileChannel fc1 = mem1.asMappableFileChannel()) {
184             int actuallyAllocatedSize1 = (int) fc1.size();
185             assertEquals(actuallyAllocatedSize0, actuallyAllocatedSize1);
186 
187             MappedByteBuffer mbb1 = fc1.map(MapMode.READ_WRITE, 0, actuallyAllocatedSize1);
188             assertEquals(0, mbb1.get());
189             assertEquals(0, mbb1.get());
190             assertEquals(0, mbb1.get());
191 
192             assertFalse(mbb1.isReadOnly());
193           }
194 
195           // read-only should work as expected
196           try (SharedMemory mem2 = SharedMemory.createOrReuseExisting(name, 3,
197               SharedMemoryOption.READ_ONLY); //
198               FileChannel fc2 = mem2.asMappableFileChannel()) {
199 
200             assertThrows(NonWritableChannelException.class, () -> fc2.map(MapMode.READ_WRITE, 0, fc2
201                 .size()));
202 
203             MappedByteBuffer mbb2 = fc2.map(MapMode.READ_ONLY, 0, fc2.size());
204             assertTrue(mbb2.isReadOnly());
205           }
206         } catch (UnsupportedOperationException e) {
207           throw new TestAbortedWithImportantMessageException(
208               MessageType.TEST_ABORTED_SHORT_INFORMATIONAL,
209               "System does not support SharedMemory#asMappableFileChannel", e);
210         }
211       }
212 
213       // We did automatically unlink the shared memory object upon close, so there isn't one now
214       assertThrows(FileNotFoundException.class, () -> SharedMemory.openExisting(name));
215     } finally {
216       SharedMemory.unlinkShared(name);
217     }
218   }

276   private void testMMapWithDuplicates(Arena arena, boolean closeArena) throws Exception {
277     boolean notMarkedAsMapped = false;
278 
279     MemorySegment ms;
280     try (SharedMemory mem = SharedMemory.createAnonymous(16)) {
281       long memSizeFC = mem.byteSize();
282       assertNotEquals(0, memSizeFC);
283 
284       ms = mem.asMappedMemorySegment(MapMode.READ_WRITE, arena, 2);
285       long memSize = ms.byteSize() / 3;
286 
287       ms.setString(0, "Hello World");
288       assertEquals("Hello World", ms.getString(0));
289       assertEquals("Hello World", ms.getString(1 * memSize));
290       assertEquals("Hello World", ms.getString(2 * memSize));
291 
292       // mmap-specific calls should succeed; these rely on Buffer having a "fd" field.
293       // However, shared memory on Windows does not have a file descriptor, therefore we
294       // need to guard this.
295       if (ms.isMapped()) {
296         ms.force(); // NOPMD.UselessPureMethodCall
297         ms.load(); // NOPMD.UselessPureMethodCall
298         ms.isLoaded(); // NOPMD.UselessPureMethodCall
299         ms.unload(); // NOPMD.UselessPureMethodCall
300       } else {
301         notMarkedAsMapped = true;
302       }
303 
304       if (closeArena) {
305         arena.close();
306       }
307 
308       // automatic unmap-upon-close relies on Buffer having a "segment" field that we can populate
309       assertNotEquals(closeArena, ms.scope().isAlive());
310 
311       if (ms.scope().isAlive()) {
312         // global scope
313         assertEquals("Hello World", ms.getString(0));
314       } else {
315         assertThrows(IllegalStateException.class, () -> ms.getString(0)); // already closed
316       }
317 
318       if (notMarkedAsMapped) {
319         throw new TestAbortedWithImportantMessageException(
320             MessageType.TEST_ABORTED_SHORT_INFORMATIONAL,
321             "SharedMemory MemorySegments are not considered #isMapped() on this platform");
322       }
323     }
324   }

402   @Test
403   @ExecutionEnvironmentRequirement(epsilonGC = Rule.PROHIBITED)
404   public void testMMapThenAccessAfterGC() throws Exception {
405     CompletableFuture<MemorySegment> ms = CompletableFuture.supplyAsync(() -> {
406       try {
407         SharedMemory mem = SharedMemory.createAnonymous(16);
408         return mem.asMappedMemorySegment(MapMode.READ_WRITE);
409       } catch (IOException e) {
410         throw new IllegalStateException(e);
411       }
412     });
413 
414     CompletableFuture<IllegalStateException> gotException = new CompletableFuture<>();
415 
416     try {
417       // Try to access MemorySegment after GC
418       new Thread(() -> {
419         for (int i = 0; i < 500; i++) { // 500 * 10ms = 5 seconds
420           try {
421             runGc();
422             ms.get().getString(0); // should throw IllegalStateException
423           } catch (IllegalStateException e) { // expected
424             gotException.complete(e);
425             break;
426           } catch (Exception e) {
427             fail(e);
428             break;
429           }
430           // if we reach this, keep trying -- there may be a slight delay between OOME and actually
431           // calling the cleaner
432           try {
433             Thread.sleep(10);
434           } catch (InterruptedException e) {
435             break;
436           }
437         }
438         gotException.complete(null); // unexpected
439 
440         assertNotNull(gotException);
441       }).start();
442 
443       // probably not necessary:
444       // GCUtil.addBrutalGCPressure(gotException::isDone);
445 
446       assertNotNull(gotException.get(5, TimeUnit.SECONDS),
447           "Should have thrown some IllegalStateException");
448     } finally {
449       gotException.complete(null); // unblock OOME thread
450     }
451   }

575   @Test
576   public void testFutexIs32Bit() throws Exception {
577     try (SharedMemory mem = SharedMemory.createAnonymous(16)) {
578       MemorySegment ms = mem.asMappedMemorySegment(MapMode.READ_WRITE).fill((byte) 0);
579 
580       ms.setAtIndex(OfInt.JAVA_INT, 1, 0xCAFEBABE);
581       assertNotEquals(0, ms.get(OfLong.JAVA_LONG, 0));
582 
583       // the bytes after the first 4 should not affect tryWait
584       long elapsed = System.currentTimeMillis();
585       try {
586         assertFalse(mem.futex(ms.asSlice(0, SharedMemory.FUTEX32_SEGMENT_SIZE)).tryWait(0,
587             FUTEX32BIT_CHECK_WAIT_TIME));
588       } catch (AssertionFailedError e) {
589         System.err.println("Spurious mutex wakeup (seen occasionally on Windows)");
590         e.printStackTrace();
591         return;
592       }
593 
594       elapsed = System.currentTimeMillis() - elapsed;
595       if (elapsed < FUTEX32BIT_CHECK_WAIT_TIME) {
596         // If tryWait returns before the expected waitTime, then we can assume that it
597         // somehow took more than the first 32-bit into consideration, and decided that the
598         // "ifValue" was not the expected one.
599         fail("The futex should have waited at least " + FUTEX32BIT_CHECK_WAIT_TIME + "ms");
600       }
601     } catch (UnsupportedOperationException e) {
602       throw new TestAbortedWithImportantMessageException(MessageType.TEST_ABORTED_SHORT_WITH_ISSUES,
603           "Futexes are not supported on your platform", e);
604     }
605   }

616   @Test
617   @ForkedVMRequirement(forkSupported = true)
618   @SuppressWarnings("PMD.ExceptionAsFlowControl")
619   public void testFutexSeparateVM() throws Exception {
620     try (SharedMemory mem = SharedMemory.createAnonymous(FORKEDVM_FUTEX_SIZE)) {
621       ForkedVM vm = new ForkedVM(FutexTestApp.class);
622 
623       CompletableFuture<@Nullable Exception> cf = CompletableFuture.supplyAsync(() -> {
624         try {
625           MemorySegment ms = mem.asMappedMemorySegment(MapMode.READ_WRITE);
626 
627           int waitRemaining = 5000;
628           long now = System.currentTimeMillis();
629 
630           try (Futex futex = mem.futex(ms.asSlice(0, 4))) {
631             if (!futex.isInterProcess()) {
632               throw new TestAbortedWithImportantMessageException(
633                   MessageType.TEST_ABORTED_SHORT_INFORMATIONAL,
634                   "On this system, SharedMemory Futexes cannot be shared between processes");
635             }
636 
637             do {
638               if (!futex.tryWait(0, waitRemaining)) {
639                 waitRemaining -= (int) (System.currentTimeMillis() - now);
640                 if (waitRemaining <= 0) {
641                   return new IOException("Timeout");
642                 }
643               } else {
644                 return null;
645               }
646             } while (true); // NOPMD
647           }
648         } catch (Exception e) {
649           return e;
650         }
651       });
652 
653       Redirect fdRedirect = FileDescriptorCast.using(mem.getFileDescriptor()).as(Redirect.class);
654       vm.setRedirectInput(fdRedirect);
655       vm.setRedirectError(Redirect.INHERIT);
656       vm.setRedirectOutput(Redirect.INHERIT);
657 
658       Process p = vm.fork();
659 
660       Exception exc;
661       try {
662         exc = cf.get(5, TimeUnit.SECONDS);
663       } catch (ExecutionException e) {
664         if (e.getCause() instanceof UnsupportedOperationException) {
665           throw new TestAbortedWithImportantMessageException(
666               MessageType.TEST_ABORTED_SHORT_WITH_ISSUES,
667               "Futexes are not supported on your platform", e);
668         }
669         throw e;
670       }
671       if (exc != null) {
672         throw exc;
673       }
674 
675       assertEquals(0, p.waitFor(), "Forked VM should have terminated successfully");
676     }

679   private void testFutex(boolean closeMem, boolean wakeUp) throws Exception {
680     try (SharedMemory mem = SharedMemory.createAnonymous(16)) {
681       MemorySegment ms = mem.asMappedMemorySegment(MapMode.READ_WRITE);
682 
683       try (Futex futex = mem.futex(ms.asSlice(4, 4).fill((byte) 0), wakeUp)) {
684         assertTrue(futex.tryWait(0xDEADBEEF, 0)); // immediately returns; value doesn't match
685 
686         try {
687           long time = System.currentTimeMillis();
688           assertFalse(futex.tryWait(0, 10)); // waits because value is 0, then times out after 10ms
689           time = System.currentTimeMillis() - time;
690           assertTrue(time >= 10, "Should have waited for at least 10 milliseconds");
691         } catch (AssertionFailedError e) {
692           System.err.println("Spurious mutex wakeup (seen occasionally on Windows)");
693           e.printStackTrace();
694         }
695 
696         AtomicBoolean tryWaitComplete = new AtomicBoolean(false);
697         Semaphore cfReady = new Semaphore(0);
698         CompletableFuture<Throwable> cf = CompletableFuture.supplyAsync(() -> {
699           try {
700             cfReady.release();
701 
702             if (closeMem) {
703               mem.close(); // effective when we take care of waking all remaining Futexes with
704                            // SharedMemory#futex(MemorySegment, true)
705             } else {
706               // We may try to wake before a wait was registered, so let's keep trying until we
707               // actually wake up some thread
708               assertTrue(futex.tryWakeWithTimeout(false, 5000, 10, () -> !tryWaitComplete.get()));
709             }
710           } catch (Throwable e) { // NOPMD.AvoidCatchingThrowable
711             return e;
712           }
713           return null;
714         });
715         cfReady.acquire(); // wait until thread is running (to improve the stability of the test)
716 
717         if (closeMem && !wakeUp) {
718           // use a shorter test interval since it's going to fail anyways
719           boolean success = futex.tryWait(0, 50);
720           tryWaitComplete.set(true);
721           assertFalse(success);
722         } else {
723           // Wait may fail because we closed the memory
724           boolean success = futex.tryWait(0, 5000) || futex.isClosed();
725           tryWaitComplete.set(true);
726           assertTrue(success);
727         }
728 
729         Throwable t = cf.get();
730         if (t != null) {
731           if (t instanceof Exception) {
732             throw (Exception) t;
733           } else {
734             throw (Error) t;
735           }
736         }
737       }
738     } catch (UnsupportedOperationException e) {
739       throw new TestAbortedWithImportantMessageException(MessageType.TEST_ABORTED_SHORT_WITH_ISSUES,
740           "Futexes are not supported on your platform", e);
741     }
742   }
